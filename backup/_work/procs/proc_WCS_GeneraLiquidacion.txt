CREATE PROCEDURE [dbo].[WCS_GeneraLiquidacion]
@Accion TINYINT,
@Fecha smalldatetime,
@Cliente INT,
@RazonSocial Varchar(50),
@Area INT,
@Total Decimal(19,5),
@GeneradaPor SMALLINT,
@Remito Varchar(20),
@FechaRemito smalldatetime,
@Factura Varchar(20),
@FechaFactura smalldatetime,
@TipoTrabajo Varchar(10),
@IDPolitica INT,
@Ordenes Ntext,
@ID INT OUTPUT
AS

SET NOCOUNT ON

DECLARE @IDoc int
DECLARE @HuboError int
DECLARE @Txt VARCHAR(500)
DECLARE @IDTrabajo int
DECLARE @Importe Decimal(19, 5)

SET @IDoc = 0 
IF DATALENGTH(@Ordenes) <> 0 
	BEGIN
		EXEC @HuboError = sp_xml_preparedocument @IDoc OUTPUT, @Ordenes
		IF @HuboError <> 0 
			BEGIN
	    				RAISERROR ('Formato inválido en XML de Ordenes', 16, 1)
					GOTO SALIR
			END
	END
ELSE
	BEGIN
		RAISERROR('No ha incluído ningún Trabajo',16, 1)
		GOTO SALIR
	END

IF @Accion = 2 
	BEGIN
		-- ALTA
		INSERT INTO Liquidaciones (	Fecha, Cliente, RazonSocial, Area, Total, GeneradaPor, Remito, FechaRemito, Factura, 
												FechaFactura, Anulado, Trabajo, IDPolitica)
		VALUES (@Fecha, @Cliente, @RazonSocial, @Area, @Total, @GeneradaPor, @Remito, @FechaRemito, @Factura, @FechaFactura, 
												0, @TipoTrabajo, @IDPolitica)
		IF @@ERROR <> 0 GOTO SALIR
		SET @ID = SCOPE_IDENTITY()
	
		INSERT INTO OrdenesLiquidadas (Liquidacion, Orden, Importe, Rechazada)
		SELECT @ID as Liquidacion, P.Orden, P.Importe, 0 as Rechazada FROM OPENXML (@idoc, '/R/I',1) WITH (Orden INT, Importe Decimal(19, 5), Rechazada bit) P
		IF @@ERROR <> 0 GOTO SALIR

		-- Generar sucesos de liquidacion...
		Declare ALiquidar Cursor LOCAL STATIC FOR Select Orden FROM OPENXML (@idoc, '/R/I',1) WITH (Orden INT, Importe Decimal(19, 5), Rechazada bit) P
		Open ALiquidar
		FETCH NEXT FROM ALiquidar INTO @IDTrabajo
		WHILE @@FETCH_STATUS = 0 AND @HuboError=0
			BEGIN
				EXEC CS_LiquidarOrden  @IDTrabajo, @Fecha, @ID, @Factura,DEFAULT,1
				IF @@ERROR <> 0 GOTO SALIR2
				FETCH NEXT FROM ALiquidar INTO @IDTrabajo
			END
		CLOSE ALiquidar
		DEALLOCATE ALiquidar
	END
ELSE
	BEGIN
		--MODIFICACION
		UPDATE Liquidaciones SET Total = @Total, Remito = @Remito, FechaRemito = @FechaRemito, Factura = @Factura, FechaFactura = @FechaFactura
		WHERE ID = @ID
		IF @@ERROR <> 0 GOTO SALIR
		-- Generar sucesos de liquidacion...
		Declare ALiquidar Cursor  LOCAL STATIC
		FOR Select Orden, Importe FROM OPENXML (@idoc, '/R/I',1) WITH (Orden INT, Importe Decimal(19, 5), Rechazada bit) P
		Open ALiquidar
		FETCH NEXT FROM ALiquidar INTO @IDTrabajo, @Importe
		WHILE @@FETCH_STATUS = 0 AND @HuboError=0
			BEGIN
				IF NOT EXISTS(SELECT * FROM OrdenesLiquidadas WHERE Liquidacion = @ID AND Orden = @IDTrabajo)
					BEGIN
						-- Agregarla a la liquidacion...
						INSERT INTO OrdenesLiquidadas (Liquidacion, Orden, Importe, Rechazada)
						SELECT @ID as Liquidacion, @IDTrabajo as Orden, @Importe, 0 as Rechazada 
						IF @@ERROR <> 0 GOTO SALIR2
						-- Suceso de Liquidacion...
						EXEC CS_LiquidarOrden @IDTrabajo, @Fecha, @ID, @Factura,DEFAULT,1
						IF @@ERROR <> 0 GOTO SALIR2
					END
				ELSE
					BEGIN
						--Actualizarla
						UPDATE OrdenesLiquidadas SET Importe = @Importe, Rechazada = 0
						WHERE Liquidacion = @ID AND  Orden = @IDTrabajo
						IF @@ERROR <> 0 GOTO SALIR2
						-- Suceso de Liquidacion...
						EXEC CS_LiquidarOrden @IDTrabajo, @Fecha, @ID, @Factura,DEFAULT,1
						IF @@ERROR <> 0 GOTO SALIR2
					END				
				FETCH NEXT FROM ALiquidar INTO @IDTrabajo, @Importe
			END
		CLOSE ALiquidar
		DEALLOCATE ALiquidar

		-- Generar sucesos de DESLIQUIDACION
		Declare ALiquidar Cursor  LOCAL STATIC FOR Select Orden FROM OrdenesLiquidadas
		WHERE Liquidacion = @ID AND Rechazada = 0 AND Orden NOT IN (SELECT Orden FROM OPENXML (@idoc, '/R/I',1) WITH (Orden INT, Importe Decimal(19, 5), Rechazada bit) )
		Open ALiquidar
		FETCH NEXT FROM ALiquidar INTO @IDTrabajo
		WHILE @@FETCH_STATUS = 0 AND @HuboError=0
			BEGIN
				UPDATE OrdenesLiquidadas SET Rechazada = 1
				WHERE Liquidacion = @ID AND  Orden = @IDTrabajo
				-- Suceso de DESLiquidacion...
				UPDATE CS_Trabajos Set Estado = 7, LiquidacionNro = 0, Liquidada = 0, FechaLiquidacion = Null, FacturaEmitida = Null
				WHERE IDTrabajo = @IDTrabajo
				SET @Txt = 'Se excluye la orden de la liquidación  Nro ' + CONVERT(VARCHAR, @ID) 
				EXEC CS_GeneraSucesoTrabajo null, 24, @IDTrabajo, @GeneradaPor, @Txt, 2
				IF @@ERROR <> 0 GOTO SALIR2
				FETCH NEXT FROM ALiquidar INTO @IDTrabajo
			END
		CLOSE ALiquidar
		DEALLOCATE ALiquidar
	END
IF @@ERROR <> 0 GOTO SALIR
GOTO FINAL

SALIR:
GOTO FINAL

SALIR2:
		CLOSE ALiquidar
		DEALLOCATE ALiquidar

FINAL:
EXEC sp_xml_removedocument @idoc 



(1 rows affected)
