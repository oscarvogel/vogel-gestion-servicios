CREATE Procedure [dbo].[CS_GetPolitica]
@ID int
AS
select A.ID, A.IDArea, A.Ppal, A.Codigo, A.Nombre, --A.Deshabilitada, 
COALESCE(A.Logo,P.Logo) as Logo, 
COALESCE(A.Autorizar,P.Autorizar) as Autorizar, 
COALESCE(A.MonedaAutorizar,P.MonedaAutorizar) as MonedaAutorizar, 
COALESCE(A.MontoAutorizar,P.MontoAutorizar) as MontoAutorizar, 
COALESCE(A.IncluirEnPartes,P.IncluirEnPartes) as IncluirEnPartes, 
COALESCE(A.ExcluirDeFactura,P.ExcluirDeFactura) as ExcluirDeFactura, 
COALESCE(A.Garantia,P.Garantia) as Garantia, 
COALESCE(A.Cerrar,P.Cerrar) as Cerrar, 
COALESCE(A.Liquidar,P.Liquidar) as Liquidar, 
COALESCE(A.Comision,P.Comision) as Comision, 
COALESCE(A.GenerarCliente,P.GenerarCliente) as GenerarCliente, 
COALESCE(A.ExigirComprobante,P.ExigirComprobante) as ExigirComprobante, 
--COALESCE(A.Comprobante,P.Comprobante) as Comprobante, 
COALESCE(A.MomentoComprobante,P.MomentoComprobante) as MomentoComprobante, 
COALESCE(A.InformarTrabajo,P.InformarTrabajo) as InformarTrabajo, 
COALESCE(A.MomentoInformar,P.MomentoInformar) as MomentoInformar, 
COALESCE(A.PedirPartes,P.PedirPartes) as PedirPartes, 
--COALESCE(A.MetodoPedirPartes,P.MetodoPedirPartes) as MetodoPedirPartes, 
COALESCE(A.FormatoPedidoPartes,P.FormatoPedidoPartes) as FormatoPedidoPartes, 
COALESCE(A.GenerarRecepcion,P.GenerarRecepcion) as GenerarRecepcion, 
COALESCE(A.GeneraMovxRecepcion,P.GeneraMovxRecepcion) as GeneraMovxRecepcion, 
COALESCE(A.DepositoRecepcion,P.DepositoRecepcion) as DepositoRecepcion, 
COALESCE(A.DevolverPartes,P.DevolverPartes) as DevolverPartes, 
COALESCE(A.GeneraMovxDevolucion,P.GeneraMovxDevolucion) as GeneraMovxDevolucion, 
COALESCE(A.DepositoDevolucion,P.DepositoDevolucion) as DepositoDevolucion, 
COALESCE(A.RecuperarPartes,P.RecuperarPartes) as RecuperarPartes, 
COALESCE(A.GeneraMovxRecupero,P.GeneraMovxRecupero) as GeneraMovxRecupero, 
COALESCE(A.DepositoRecupero,P.DepositoRecupero) as DepositoRecupero, 
COALESCE(A.SoloProductosMarca,P.SoloProductosMarca) as SoloProductosMarca, 
COALESCE(A.ValidarContraCatalogo,P.ValidarContraCatalogo) as ValidarContraCatalogo, 
COALESCE(A.CatalogoDePropiedades,P.CatalogoDePropiedades) as CatalogoDePropiedades, 
COALESCE(A.PartesProvistas,P.PartesProvistas) as PartesProvistas, 
COALESCE(A.PermiteProvisionTemporal,P.PermiteProvisionTemporal) as PermiteProvisionTemporal, 
--COALESCE(A.PermiteProvisionDefinitiva,P.PermiteProvisionDefinitiva) as PermiteProvisionDefinitiva, 
COALESCE(A.FormatoPedido,P.FormatoPedido) as FormatoPedido, 
COALESCE(A.AutolistarPedido,P.AutolistarPedido) as AutolistarPedido, 
COALESCE(A.CopiasClientePedido,P.CopiasClientePedido) as CopiasClientePedido, 
COALESCE(A.CopiasInternasPedido,P.CopiasInternasPedido) as CopiasInternasPedido, 
COALESCE(A.FormatoRecepcion,P.FormatoRecepcion) as FormatoRecepcion, 
COALESCE(A.AutolistarRecepcion,P.AutolistarRecepcion) as AutolistarRecepcion, 
COALESCE(A.DiasDiagnostico,P.DiasDiagnostico) as DiasDiagnostico, 
COALESCE(A.HorarioAtencion,P.HorarioAtencion) as HorarioAtencion, 
COALESCE(A.CopiasClienteRecepcion,P.CopiasClienteRecepcion) as CopiasClienteRecepcion, 
COALESCE(A.CopiasInternasRecepcion,P.CopiasInternasRecepcion) as CopiasInternasRecepcion, 
COALESCE(A.PieRecepcion,P.PieRecepcion) as PieRecepcion, 
COALESCE(A.FormatoOrden,P.FormatoOrden) as FormatoOrden, 
COALESCE(A.CopiasOrden,P.CopiasOrden) as CopiasOrden, 
COALESCE(A.FormatoOrdenCliente,P.FormatoOrdenCliente) as FormatoOrdenCliente, 
COALESCE(A.CopiasOrdenCliente,P.CopiasOrdenCliente) as CopiasOrdenCliente, 
COALESCE(A.FormatoLiquidacion,P.FormatoLiquidacion) as FormatoLiquidacion, 
COALESCE(A.ExportadorLiquidacion,P.ExportadorLiquidacion) as ExportadorLiquidacion, 
COALESCE(A.SistemaCodificacion,P.SistemaCodificacion) as SistemaCodificacion,
COALESCE(A.PiePedido,P.PiePedido) as PiePedido,
COALESCE(A.PieRetiro,P.PieRetiro) as PieRetiro
From cs_politicasxarea A
LEFT JOIN cs_politicasxarea P ON P.IDArea = A.IDArea AND P.Ppal <> 0
Where A.ID = @ID

(1 rows affected)
