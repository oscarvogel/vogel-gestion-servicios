CREATE VIEW  [dbo].[CS_OrdenesLiquidacionTecnicoFalabella]
AS
SELECT T.IDTrabajo, T.IDPolitica, T.NroOtorgado, T.FechaAlta, T.FechaLista as FechaCerrada, T.Cliente, CONVERT(Varchar(150), T.TrabajoRealizado) as TrabajoRealizado, 
Pr.Provincia, T.Localidad as Localidad, OL.Importe, T.Modelo, T.Equipo, T.Comprobante, T.NroDocumento,
COALESCE(I.ComisionServicios, 0) as ComisionServicios,
COALESCE(ComisionOtros, 0) + COALESCE(I.ComisionProductos, 0) +  COALESCE(I.ComisionPartes, 0) as ComisionOtros,
L.ID, L.Fecha, P.NombreCompleto,
Ag.ID as IDAgrupacion, COALESCE(Ag.Descripcion, 'Sin Agrupación') as Agrupacion, M.Codigo as IDMarca, COALESCE(M.Descripcion, 'Sin Marca') as Marca, COALESCE(C.Cuenta, 'Sin Cuenta') as Cuenta,
L.Tecnico, OL.Rechazada, OL.Liquidacion,
Pa.Descripcion as DescripcionServicio,
CASE WHEN Pa.Tipo = 2 THEN  COALESCE(Cp.Comision, I.ComisionServicios, 0) ELSE 0 END as CostoServicios,
CASE WHEN Pa.Tipo <> 2 THEN COALESCE(Cp.Comision, COALESCE(ComisionOtros, 0) + COALESCE(I.ComisionProductos, 0) +  COALESCE(I.ComisionPartes, 0), 0) ELSE 0 END as CostoOtros
FROM CS_OrdenesLiquidadasTecnicos OL WITH (NOLOCK) 
INNER JOIN CS_Trabajos T WITH (NOLOCK) ON T.IDTrabajo = OL.Orden
INNER JOIN CS_LiquidacionesTecnicos L WITH (NOLOCK) ON L.ID = OL.Liquidacion
INNER JOIN Intervienen I WITH (NOLOCK) ON I.Liquidacion = L.ID AND I.Orden = T.IDTrabajo AND I.Persona = OL.Tecnico
INNER JOIN CS_PoliticasxArea P WITH (NOLOCK) ON P.ID = T.IDPolitica
LEFT JOIN CC_Articulos A WITH (NOLOCK) ON A.Codigo = T.Modelo
LEFT JOIN CC_Marcas M WITH (NOLOCK) ON M.Codigo = A.Marca
LEFT JOIN CC_Agrupaciones Ag WITH (NOLOCK) ON Ag.ID =  A.Agrupacion
LEFT JOIN CS_CuentasAgrupacionMarca C WITH (NOLOCK) ON C.IDAgrupacion = Ag.ID AND C.IDMarca = A.Marca
LEFT JOIN CC_Provincias Pr WITH (NOLOCK) ON Pr.ID = T.Provincia
LEFT JOIN CS_Partes Pa WITH (NOLOCK) ON Pa.NroOrden = T.IDTrabajo AND Pa.Tipo >= 2
LEFT JOIN CS_ComisionPartexInterviene Cp WITH (NOLOCK) ON Cp.ID = Pa.ID AND Cp.IDUsuario = OL.Tecnico
WHERE OL.Rechazada = 0


(1 rows affected)
